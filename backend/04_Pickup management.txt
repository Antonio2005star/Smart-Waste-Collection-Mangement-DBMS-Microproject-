-- PURPOSE:
-- This feature identifies bins requiring collection,
-- displays pickup details and completes bin pickups
-- while enforcing the route stop sequence.
--
-- ============================================================


-- ============================================================
-- SQL QUERY 1: FIND BINS REQUIRING PICKUP
-- DESCRIPTION:
-- Displays bins whose fill level has reached 80% or more.
-- ============================================================

COLUMN BIN_ID FORMAT 9999
COLUMN FILL_LEVEL_PERCENT FORMAT 990.0

SELECT
    bin_id,
    fill_level_percent

FROM Bins

WHERE fill_level_percent >= 80

ORDER BY fill_level_percent DESC;


-- ============================================================
-- SQL QUERY 2: VIEW PICKUP DETAILS
-- DESCRIPTION:
-- Displays schedule, route, bin, location, zone,
-- driver, vehicle and stop sequence.
-- ============================================================

COLUMN SCHEDULE_ID FORMAT 9999
COLUMN BIN_ID FORMAT 9999
COLUMN STREET_ADDRESS FORMAT A25
COLUMN ZONE_NAME FORMAT A25
COLUMN FILL_LEVEL_PERCENT FORMAT 990.0
COLUMN ROUTE_ID FORMAT 9999
COLUMN ROUTE_NAME FORMAT A25
COLUMN DRIVER_NAME FORMAT A25
COLUMN VEHICLE FORMAT A15
COLUMN STOP_SEQUENCE FORMAT 999

SELECT
    s.schedule_id,
    b.bin_id,
    l.street_address,
    z.zone_name,
    b.fill_level_percent,
    r.route_id,
    r.route_name,
    d.first_name || ' ' || d.last_name AS driver_name,
    v.registration_number AS vehicle,
    rb.stop_sequence

FROM Schedules s

JOIN Routes r
    ON s.route_id = r.route_id

JOIN Route_Bins rb
    ON r.route_id = rb.route_id

JOIN Bins b
    ON rb.bin_id = b.bin_id

JOIN Locations l
    ON b.location_id = l.location_id

JOIN Zones z
    ON l.zone_id = z.zone_id

LEFT JOIN Drivers d
    ON r.driver_id = d.driver_id

LEFT JOIN Vehicles v
    ON r.vehicle_id = v.vehicle_id

WHERE b.fill_level_percent >= 80

ORDER BY
    r.route_id,
    rb.stop_sequence;


-- ============================================================
-- SQL QUERY 3: VIEW ROUTE COLLECTION STATUS
-- DESCRIPTION:
-- Displays the collection status of every stop
-- in each route.
-- ============================================================

COLUMN ROUTE_ID FORMAT 9999
COLUMN BIN_ID FORMAT 9999
COLUMN STOP_SEQUENCE FORMAT 999
COLUMN COLLECTION_STATUS FORMAT A20

SELECT
    route_id,
    bin_id,
    stop_sequence,
    collection_status

FROM Route_Bins

ORDER BY
    route_id,
    stop_sequence;


-- ============================================================
-- SQL QUERY 4: VIEW PENDING PICKUPS
-- DESCRIPTION:
-- Displays bins that are pending collection
-- and have reached the pickup threshold.
-- ============================================================

COLUMN ROUTE_ID FORMAT 9999
COLUMN BIN_ID FORMAT 9999
COLUMN STOP_SEQUENCE FORMAT 999
COLUMN COLLECTION_STATUS FORMAT A20
COLUMN FILL_LEVEL_PERCENT FORMAT 990.0

SELECT
    rb.route_id,
    rb.bin_id,
    rb.stop_sequence,
    rb.collection_status,
    b.fill_level_percent

FROM Route_Bins rb

JOIN Bins b
    ON rb.bin_id = b.bin_id

WHERE rb.collection_status = 'Pending'
  AND b.fill_level_percent >= 80

ORDER BY
    rb.route_id,
    rb.stop_sequence;


-- ============================================================
-- PL/SQL: PROCEDURE TO COMPLETE BIN PICKUP
-- DESCRIPTION:
-- Checks the schedule and route, enforces stop sequence,
-- marks the bin as collected, empties the bin and updates
-- the schedule status.
-- ============================================================

CREATE OR REPLACE PROCEDURE COMPLETE_BIN_PICKUP (
    P_SCHEDULE_ID IN NUMBER,
    P_ROUTE_ID    IN NUMBER,
    P_BIN_ID      IN NUMBER
)
IS

    V_SEQUENCE          NUMBER;
    V_PREVIOUS_STATUS   VARCHAR2(20);
    V_CURRENT_STATUS    VARCHAR2(20);
    V_TOTAL             NUMBER;
    V_COMPLETED         NUMBER;
    V_ROUTE_ID          NUMBER;

BEGIN

    -- --------------------------------------------------------
    -- Check that the schedule exists and belongs to the route
    -- --------------------------------------------------------

    SELECT ROUTE_ID
    INTO V_ROUTE_ID

    FROM Schedules

    WHERE SCHEDULE_ID = P_SCHEDULE_ID;


    IF V_ROUTE_ID <> P_ROUTE_ID THEN

        RAISE_APPLICATION_ERROR(
            -20001,
            'Invalid schedule and route.'
        );

    END IF;


    -- --------------------------------------------------------
    -- Get stop sequence and current collection status
    -- --------------------------------------------------------

    SELECT
        STOP_SEQUENCE,
        COLLECTION_STATUS

    INTO
        V_SEQUENCE,
        V_CURRENT_STATUS

    FROM Route_Bins

    WHERE ROUTE_ID = P_ROUTE_ID
      AND BIN_ID = P_BIN_ID;


    -- --------------------------------------------------------
    -- Check whether the bin is already collected
    -- --------------------------------------------------------

    IF V_CURRENT_STATUS = 'Collected' THEN

        RAISE_APPLICATION_ERROR(
            -20003,
            'This bin has already been collected.'
        );

    END IF;


    -- --------------------------------------------------------
    -- Check previous stop
    -- --------------------------------------------------------

    IF V_SEQUENCE > 1 THEN

        SELECT COLLECTION_STATUS
        INTO V_PREVIOUS_STATUS

        FROM Route_Bins

        WHERE ROUTE_ID = P_ROUTE_ID
          AND STOP_SEQUENCE = V_SEQUENCE - 1;


        IF NVL(V_PREVIOUS_STATUS, 'Pending') <> 'Collected' THEN

            RAISE_APPLICATION_ERROR(
                -20002,
                'Complete the previous stop first.'
            );

        END IF;

    END IF;


    -- --------------------------------------------------------
    -- Mark pickup as completed
    -- --------------------------------------------------------

    UPDATE Route_Bins

    SET COLLECTION_STATUS = 'Collected'

    WHERE ROUTE_ID = P_ROUTE_ID
      AND BIN_ID = P_BIN_ID;


    -- --------------------------------------------------------
    -- Empty the collected bin
    -- --------------------------------------------------------

    UPDATE Bins

    SET FILL_LEVEL_PERCENT = 0

    WHERE BIN_ID = P_BIN_ID;


    -- --------------------------------------------------------
    -- Count total stops
    -- --------------------------------------------------------

    SELECT COUNT(*)
    INTO V_TOTAL

    FROM Route_Bins

    WHERE ROUTE_ID = P_ROUTE_ID;


    -- --------------------------------------------------------
    -- Count completed stops
    -- --------------------------------------------------------

    SELECT COUNT(*)
    INTO V_COMPLETED

    FROM Route_Bins

    WHERE ROUTE_ID = P_ROUTE_ID
      AND COLLECTION_STATUS = 'Collected';


    -- --------------------------------------------------------
    -- Update schedule status
    -- --------------------------------------------------------

    IF V_TOTAL > 0
       AND V_COMPLETED = V_TOTAL THEN

        UPDATE Schedules

        SET STATUS = 'Completed'

        WHERE SCHEDULE_ID = P_SCHEDULE_ID;


    ELSIF V_COMPLETED > 0 THEN

        UPDATE Schedules

        SET STATUS = 'In Progress'

        WHERE SCHEDULE_ID = P_SCHEDULE_ID;


    ELSE

        UPDATE Schedules

        SET STATUS = 'Scheduled'

        WHERE SCHEDULE_ID = P_SCHEDULE_ID;

    END IF;


    -- --------------------------------------------------------
    -- Save changes
    -- --------------------------------------------------------

    COMMIT;


    -- --------------------------------------------------------
    -- Display success message
    -- --------------------------------------------------------

    DBMS_OUTPUT.PUT_LINE(
        '===== BIN PICKUP COMPLETED ====='
    );

    DBMS_OUTPUT.PUT_LINE(
        'Schedule ID     : ' || P_SCHEDULE_ID
    );

    DBMS_OUTPUT.PUT_LINE(
        'Route ID        : ' || P_ROUTE_ID
    );

    DBMS_OUTPUT.PUT_LINE(
        'Bin ID          : ' || P_BIN_ID
    );

    DBMS_OUTPUT.PUT_LINE(
        'Stop Sequence   : ' || V_SEQUENCE
    );

    DBMS_OUTPUT.PUT_LINE(
        'Completed Stops : ' ||
        V_COMPLETED || '/' || V_TOTAL
    );


    IF V_TOTAL > 0
       AND V_COMPLETED = V_TOTAL THEN

        DBMS_OUTPUT.PUT_LINE(
            'Schedule Status : Completed'
        );

    ELSIF V_COMPLETED > 0 THEN

        DBMS_OUTPUT.PUT_LINE(
            'Schedule Status : In Progress'
        );

    ELSE

        DBMS_OUTPUT.PUT_LINE(
            'Schedule Status : Scheduled'
        );

    END IF;


EXCEPTION

    WHEN NO_DATA_FOUND THEN

        ROLLBACK;

        DBMS_OUTPUT.PUT_LINE(
            'Invalid schedule, route or bin details.'
        );

    WHEN OTHERS THEN

        ROLLBACK;

        DBMS_OUTPUT.PUT_LINE(
            'Error: ' || SQLERRM
        );

END;
/

SHOW ERRORS;
