SET PAGESIZE 50
SET TAB OFF
SET TRIMSPOOL ON
SET SERVEROUTPUT ON
SET SQLBLANKLINES ON


-- ============================================================
-- SQL QUERY 1: ROUTE OVERVIEW
-- DESCRIPTION:
-- Displays routes with their zone, assigned driver,
-- assigned vehicle and number of stops.
-- ============================================================

COLUMN ROUTE_ID FORMAT 9999
COLUMN ROUTE_NAME FORMAT A25
COLUMN ZONE_NAME FORMAT A25
COLUMN DRIVER_NAME FORMAT A25
COLUMN VEHICLE FORMAT A15
COLUMN STOP_COUNT FORMAT 999

SELECT
    r.route_id,
    r.route_name,
    z.zone_name,
    d.first_name || ' ' || d.last_name AS driver_name,
    v.registration_number AS vehicle,
    COUNT(rb.bin_id) AS stop_count

FROM Routes r

JOIN Zones z
    ON r.zone_id = z.zone_id

LEFT JOIN Drivers d
    ON r.driver_id = d.driver_id

LEFT JOIN Vehicles v
    ON r.vehicle_id = v.vehicle_id

LEFT JOIN Route_Bins rb
    ON r.route_id = rb.route_id

GROUP BY
    r.route_id,
    r.route_name,
    z.zone_name,
    d.first_name,
    d.last_name,
    v.registration_number

ORDER BY r.route_id;


-- ============================================================
-- SQL QUERY 2: DISPLAY ROUTE STOPS
-- DESCRIPTION:
-- Displays the bins assigned to a route in their
-- defined stop sequence.
-- ============================================================

COLUMN STOP_SEQUENCE FORMAT 999
COLUMN BIN_ID FORMAT 9999
COLUMN STREET_ADDRESS FORMAT A25
COLUMN ZONE_NAME FORMAT A25
COLUMN FILL_LEVEL_PERCENT FORMAT 990.0
COLUMN PRIORITY FORMAT A10

SELECT
    rb.stop_sequence,
    b.bin_id,
    l.street_address,
    z.zone_name,
    b.fill_level_percent,

    CASE
        WHEN b.fill_level_percent >= 90 THEN 'Critical'
        WHEN b.fill_level_percent >= 80 THEN 'Warning'
        ELSE 'Normal'
    END AS priority

FROM Route_Bins rb

JOIN Bins b
    ON rb.bin_id = b.bin_id

JOIN Locations l
    ON b.location_id = l.location_id

JOIN Zones z
    ON l.zone_id = z.zone_id

WHERE rb.route_id = 100

ORDER BY rb.stop_sequence;


-- ============================================================
-- SQL QUERY 3: ASSIGN DRIVER TO ROUTE
-- DESCRIPTION:
-- Assigns a driver to a selected route.
-- ============================================================

UPDATE Routes
SET driver_id = 1
WHERE route_id = 400;

COMMIT;


-- ============================================================
-- SQL QUERY 4: ASSIGN VEHICLE TO ROUTE
-- DESCRIPTION:
-- Assigns a vehicle to a selected route.
-- ============================================================

UPDATE Routes
SET vehicle_id = 3
WHERE route_id = 400;

COMMIT;


-- ============================================================
-- SQL QUERY 5: CHECK DRIVER AND VEHICLE ASSIGNMENTS
-- DESCRIPTION:
-- Displays the current driver and vehicle assigned
-- to each route.
-- ============================================================

COLUMN ROUTE_ID FORMAT 9999
COLUMN ROUTE_NAME FORMAT A25
COLUMN DRIVER FORMAT A25
COLUMN VEHICLE FORMAT A15

SELECT
    r.route_id,
    r.route_name,
    d.first_name || ' ' || d.last_name AS driver,
    v.registration_number AS vehicle

FROM Routes r

LEFT JOIN Drivers d
    ON r.driver_id = d.driver_id

LEFT JOIN Vehicles v
    ON r.vehicle_id = v.vehicle_id

ORDER BY r.route_id;


-- ============================================================
-- SQL QUERY 6: DISPLAY ALL ROUTES WITH THEIR STOPS
-- DESCRIPTION:
-- Displays every route along with its assigned bins
-- and stop sequence.
-- ============================================================

COLUMN ROUTE_ID FORMAT 9999
COLUMN ROUTE_NAME FORMAT A25
COLUMN STOP_SEQUENCE FORMAT 999
COLUMN BIN_ID FORMAT 9999
COLUMN STREET_ADDRESS FORMAT A25
COLUMN ZONE_NAME FORMAT A25
COLUMN FILL_LEVEL_PERCENT FORMAT 990.0

SELECT
    r.route_id,
    r.route_name,
    rb.stop_sequence,
    b.bin_id,
    l.street_address,
    z.zone_name,
    b.fill_level_percent

FROM Routes r

JOIN Route_Bins rb
    ON r.route_id = rb.route_id

JOIN Bins b
    ON rb.bin_id = b.bin_id

JOIN Locations l
    ON b.location_id = l.location_id

JOIN Zones z
    ON l.zone_id = z.zone_id

ORDER BY
    r.route_id,
    rb.stop_sequence;


-- ============================================================
-- SQL QUERY 7: CHECK ROUTE STOP SEQUENCE
-- DESCRIPTION:
-- Displays the collection status of each stop
-- in a selected route.
-- ============================================================

COLUMN ROUTE_ID FORMAT 9999
COLUMN BIN_ID FORMAT 9999
COLUMN STOP_SEQUENCE FORMAT 999
COLUMN COLLECTION_STATUS FORMAT A20

SELECT
    route_id,
    bin_id,
    stop_sequence,
    collection_status

FROM Route_Bins

WHERE route_id = 100

ORDER BY stop_sequence;


-- ============================================================
-- SQL QUERY 8: CALCULATE ROUTE PROGRESS
-- DESCRIPTION:
-- Counts the total stops and completed stops
-- for a selected route.
-- ============================================================

COLUMN ROUTE_ID FORMAT 9999
COLUMN TOTAL_STOPS FORMAT 999
COLUMN COMPLETED_STOPS FORMAT 999

SELECT
    route_id,
    COUNT(*) AS total_stops,

    SUM(
        CASE
            WHEN collection_status = 'Collected'
            THEN 1
            ELSE 0
        END
    ) AS completed_stops

FROM Route_Bins

WHERE route_id = 100

GROUP BY route_id;


-- ============================================================
-- SQL QUERY 9: CHECK PREVIOUS STOP STATUS
-- DESCRIPTION:
-- Checks whether a previous route stop has been
-- completed before continuing to the next stop.
-- ============================================================

COLUMN COLLECTION_STATUS FORMAT A20

SELECT
    collection_status

FROM Route_Bins

WHERE route_id = 100
AND stop_sequence = 2;


-- ============================================================
-- PL/SQL 1: FUNCTION TO DETERMINE BIN PRIORITY
-- DESCRIPTION:
-- Returns the priority of a bin based on its
-- current fill level.
--
-- 90% or above -> Critical
-- 80% to 89%   -> Warning
-- Below 80%    -> Normal
-- ============================================================

CREATE OR REPLACE FUNCTION GET_BIN_PRIORITY (
    P_FILL_LEVEL NUMBER
)
RETURN VARCHAR2
IS
BEGIN

    IF P_FILL_LEVEL >= 90 THEN

        RETURN 'Critical';

    ELSIF P_FILL_LEVEL >= 80 THEN

        RETURN 'Warning';

    ELSE

        RETURN 'Normal';

    END IF;

END;
/

-- ============================================================
-- PL/SQL 2: PROCEDURE TO COMPLETE BIN PICKUP
-- DESCRIPTION:
-- Ensures that route stops are completed in sequence.
-- It marks the selected stop as collected, empties the bin,
-- and updates the schedule status.
-- ============================================================

CREATE OR REPLACE PROCEDURE COMPLETE_BIN_PICKUP (
    P_SCHEDULE_ID IN NUMBER,
    P_ROUTE_ID    IN NUMBER,
    P_BIN_ID      IN NUMBER
)
IS

    V_SEQUENCE         NUMBER;
    V_PREVIOUS_STATUS  VARCHAR2(20);
    V_CURRENT_STATUS   VARCHAR2(20);
    V_TOTAL            NUMBER;
    V_COMPLETED        NUMBER;
    V_ROUTE            NUMBER;

BEGIN

    -- --------------------------------------------------------
    -- Check that the schedule belongs to the selected route
    -- --------------------------------------------------------

    SELECT ROUTE_ID
    INTO V_ROUTE

    FROM SCHEDULES

    WHERE SCHEDULE_ID = P_SCHEDULE_ID;


    IF V_ROUTE <> P_ROUTE_ID THEN

        RAISE_APPLICATION_ERROR(
            -20001,
            'Invalid schedule and route.'
        );

    END IF;


    -- --------------------------------------------------------
    -- Get the stop sequence and current status
    -- --------------------------------------------------------

    SELECT
        STOP_SEQUENCE,
        COLLECTION_STATUS

    INTO
        V_SEQUENCE,
        V_CURRENT_STATUS

    FROM ROUTE_BINS

    WHERE ROUTE_ID = P_ROUTE_ID
      AND BIN_ID = P_BIN_ID;


    -- --------------------------------------------------------
    -- Check whether the bin is already collected
    -- --------------------------------------------------------

    IF V_CURRENT_STATUS = 'Collected' THEN

        RAISE_APPLICATION_ERROR(
            -20003,
            'This bin has already been collected.'
        );

    END IF;


    -- --------------------------------------------------------
    -- Check previous stop
    -- --------------------------------------------------------

    IF V_SEQUENCE > 1 THEN

        SELECT COLLECTION_STATUS
        INTO V_PREVIOUS_STATUS

        FROM ROUTE_BINS

        WHERE ROUTE_ID = P_ROUTE_ID
          AND STOP_SEQUENCE = V_SEQUENCE - 1;


        IF NVL(V_PREVIOUS_STATUS, 'Pending') <> 'Collected' THEN

            RAISE_APPLICATION_ERROR(
                -20002,
                'Complete the previous stop first.'
            );

        END IF;

    END IF;


    -- --------------------------------------------------------
    -- Mark the selected stop as collected
    -- --------------------------------------------------------

    UPDATE ROUTE_BINS

    SET COLLECTION_STATUS = 'Collected'

    WHERE ROUTE_ID = P_ROUTE_ID
      AND BIN_ID = P_BIN_ID;


    -- --------------------------------------------------------
    -- Empty the collected bin
    -- --------------------------------------------------------

    UPDATE BINS

    SET FILL_LEVEL_PERCENT = 0

    WHERE BIN_ID = P_BIN_ID;


    -- --------------------------------------------------------
    -- Count total route stops
    -- --------------------------------------------------------

    SELECT COUNT(*)
    INTO V_TOTAL

    FROM ROUTE_BINS

    WHERE ROUTE_ID = P_ROUTE_ID;


    -- --------------------------------------------------------
    -- Count completed route stops
    -- --------------------------------------------------------

    SELECT COUNT(*)
    INTO V_COMPLETED

    FROM ROUTE_BINS

    WHERE ROUTE_ID = P_ROUTE_ID
      AND COLLECTION_STATUS = 'Collected';


    -- --------------------------------------------------------
    -- Update schedule status
    -- --------------------------------------------------------

    IF V_COMPLETED = V_TOTAL THEN

        UPDATE SCHEDULES

        SET STATUS = 'Completed'

        WHERE SCHEDULE_ID = P_SCHEDULE_ID;

    ELSE

        UPDATE SCHEDULES

        SET STATUS = 'In Progress'

        WHERE SCHEDULE_ID = P_SCHEDULE_ID
          AND STATUS = 'Scheduled';

    END IF;


    -- --------------------------------------------------------
    -- Save changes
    -- --------------------------------------------------------

    COMMIT;


    DBMS_OUTPUT.PUT_LINE(
        'Bin ' || P_BIN_ID ||
        ' collected successfully.'
    );

    DBMS_OUTPUT.PUT_LINE(
        'Route ID: ' || P_ROUTE_ID
    );

    DBMS_OUTPUT.PUT_LINE(
        'Stop Sequence: ' || V_SEQUENCE
    );

    DBMS_OUTPUT.PUT_LINE(
        'Completed Stops: ' ||
        V_COMPLETED || '/' || V_TOTAL
    );


EXCEPTION

    WHEN NO_DATA_FOUND THEN

        ROLLBACK;

        DBMS_OUTPUT.PUT_LINE(
            'Invalid schedule, route or bin details.'
        );

    WHEN OTHERS THEN

        ROLLBACK;

        DBMS_OUTPUT.PUT_LINE(
            'Error: ' || SQLERRM
        );

END;
/

SHOW ERRORS;

    );

END;
