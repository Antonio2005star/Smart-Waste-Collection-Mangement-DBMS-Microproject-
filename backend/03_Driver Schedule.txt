-- ============================================================
-- PURPOSE
-- ============================================================
-- This feature displays driver schedules, including
-- schedule date, route details, vehicle information,
-- timings, and schedule status.
--
-- The queries display all schedules unless specifically
-- mentioned otherwise.
--
-- ============================================================


-- ============================================================
-- SQL QUERY 1: VIEW ALL DRIVER SCHEDULES
-- DESCRIPTION:
-- Displays all driver schedules with schedule date,
-- route, assigned driver, vehicle, timings, and status.
-- ============================================================

COLUMN SCHEDULE_ID FORMAT 9999
COLUMN SCHEDULE_DATE FORMAT A12
COLUMN ROUTE_NAME FORMAT A25
COLUMN DRIVER_NAME FORMAT A25
COLUMN VEHICLE FORMAT A15
COLUMN START_TIME FORMAT A12
COLUMN END_TIME FORMAT A12
COLUMN STATUS FORMAT A15

SELECT
    s.schedule_id,
    TO_CHAR(s.scheduled_date, 'DD-MM-YYYY') AS schedule_date,
    r.route_name,

    d.first_name || ' ' || d.last_name
        AS driver_name,

    v.registration_number
        AS vehicle,

    s.start_time,
    s.end_time,
    s.status

FROM Schedules s

JOIN Routes r
    ON s.route_id = r.route_id

LEFT JOIN Drivers d
    ON r.driver_id = d.driver_id

LEFT JOIN Vehicles v
    ON r.vehicle_id = v.vehicle_id

ORDER BY
    s.scheduled_date,
    s.start_time;


-- ============================================================
-- SQL QUERY 2: VIEW ALL SCHEDULES BY DRIVER
-- DESCRIPTION:
-- Displays each driver's assigned route and schedule
-- date, timings, and status.
-- ============================================================

COLUMN DRIVER_ID FORMAT 9999
COLUMN DRIVER_NAME FORMAT A25
COLUMN SCHEDULE_DATE FORMAT A12
COLUMN ROUTE_ID FORMAT 9999
COLUMN ROUTE_NAME FORMAT A25
COLUMN START_TIME FORMAT A12
COLUMN END_TIME FORMAT A12
COLUMN STATUS FORMAT A15

SELECT
    d.driver_id,
    d.first_name || ' ' || d.last_name AS driver_name,
    TO_CHAR(s.scheduled_date, 'DD-MM-YYYY') AS schedule_date,
    r.route_id,
    r.route_name,
    s.start_time,
    s.end_time,
    s.status

FROM Schedules s

JOIN Routes r
    ON s.route_id = r.route_id

JOIN Drivers d
    ON r.driver_id = d.driver_id

ORDER BY
    driver_name,
    s.scheduled_date,
    s.start_time;


-- ============================================================
-- SQL QUERY 3: VIEW DRIVER SCHEDULE WITH VEHICLE DETAILS
-- DESCRIPTION:
-- Displays schedule date, driver, route, vehicle,
-- timing, and schedule status.
-- ============================================================

COLUMN SCHEDULE_ID FORMAT 9999
COLUMN SCHEDULE_DATE FORMAT A12
COLUMN DRIVER_NAME FORMAT A25
COLUMN ROUTE_NAME FORMAT A25
COLUMN VEHICLE_ID FORMAT 9999
COLUMN VEHICLE FORMAT A15
COLUMN START_TIME FORMAT A12
COLUMN END_TIME FORMAT A12
COLUMN STATUS FORMAT A15

SELECT
    s.schedule_id,
    TO_CHAR(s.scheduled_date, 'DD-MM-YYYY') AS schedule_date,

    d.first_name || ' ' || d.last_name
        AS driver_name,

    r.route_name,
    v.vehicle_id,
    v.registration_number AS vehicle,
    s.start_time,
    s.end_time,
    s.status

FROM Schedules s

JOIN Routes r
    ON s.route_id = r.route_id

LEFT JOIN Drivers d
    ON r.driver_id = d.driver_id

LEFT JOIN Vehicles v
    ON r.vehicle_id = v.vehicle_id

ORDER BY
    s.scheduled_date,
    s.start_time;


-- ============================================================
-- SQL QUERY 4: CHECK DRIVER SCHEDULE STATUS
-- DESCRIPTION:
-- Displays the schedule date, driver, route, and
-- current status of each driver schedule.
-- ============================================================

COLUMN SCHEDULE_ID FORMAT 9999
COLUMN SCHEDULE_DATE FORMAT A12
COLUMN DRIVER_NAME FORMAT A25
COLUMN ROUTE_NAME FORMAT A25
COLUMN STATUS FORMAT A15

SELECT
    s.schedule_id,
    TO_CHAR(s.scheduled_date, 'DD-MM-YYYY') AS schedule_date,

    d.first_name || ' ' || d.last_name
        AS driver_name,

    r.route_name,
    s.status

FROM Schedules s

JOIN Routes r
    ON s.route_id = r.route_id

LEFT JOIN Drivers d
    ON r.driver_id = d.driver_id

ORDER BY
    s.status,
    s.scheduled_date,
    d.first_name;


-- ============================================================
-- SQL QUERY 5: COUNT SCHEDULES BY STATUS
-- DESCRIPTION:
-- Counts all driver schedules according to their status.
-- ============================================================

COLUMN STATUS FORMAT A15
COLUMN TOTAL_SCHEDULES FORMAT 999

SELECT
    status,
    COUNT(*) AS total_schedules

FROM Schedules

GROUP BY status

ORDER BY status;


-- ============================================================
-- SQL QUERY 6: VIEW COMPLETE SCHEDULE INFORMATION
-- DESCRIPTION:
-- Displays complete schedule details including
-- schedule date, route, driver, vehicle, timings,
-- and current status.
-- ============================================================

COLUMN SCHEDULE_ID FORMAT 9999
COLUMN SCHEDULE_DATE FORMAT A12
COLUMN ROUTE_ID FORMAT 9999
COLUMN ROUTE_NAME FORMAT A25
COLUMN DRIVER_NAME FORMAT A25
COLUMN VEHICLE FORMAT A15
COLUMN START_TIME FORMAT A12
COLUMN END_TIME FORMAT A12
COLUMN STATUS FORMAT A15

SELECT
    s.schedule_id,
    TO_CHAR(s.scheduled_date, 'DD-MM-YYYY') AS schedule_date,
    r.route_id,
    r.route_name,

    d.first_name || ' ' || d.last_name
        AS driver_name,

    v.registration_number AS vehicle,
    s.start_time,
    s.end_time,
    s.status

FROM Schedules s

JOIN Routes r
    ON s.route_id = r.route_id

LEFT JOIN Drivers d
    ON r.driver_id = d.driver_id

LEFT JOIN Vehicles v
    ON r.vehicle_id = v.vehicle_id

ORDER BY
    s.scheduled_date,
    s.start_time;
