SET SERVEROUTPUT ON;

-- 1. Tables
DROP TABLE Fill_Level_History CASCADE CONSTRAINTS;
DROP TABLE Bins CASCADE CONSTRAINTS;

CREATE TABLE Bins (
    bin_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    bin_type VARCHAR2(50),      
    area_name VARCHAR2(100),    
    location_address VARCHAR2(255), -- ADDED THIS: Needed for Reports and Master Test
    fill_level NUMBER DEFAULT 0 CHECK (fill_level BETWEEN 0 AND 100),
    collection_status VARCHAR2(20) DEFAULT 'Normal'
);

CREATE TABLE Fill_Level_History (
    history_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    bin_id NUMBER REFERENCES Bins(bin_id) ON DELETE CASCADE, -- ADDED: ON DELETE CASCADE for easier testing
    recorded_level NUMBER,
    reading_timestamp TIMESTAMP DEFAULT SYSTIMESTAMP
);

-- 2. Function
CREATE OR REPLACE FUNCTION Get_Fill_Status_Label(p_level IN NUMBER) 
RETURN VARCHAR2 AS
BEGIN
    IF p_level >= 90 THEN RETURN 'Full';
    ELSIF p_level >= 75 THEN RETURN 'Nearly Full';
    ELSE RETURN 'Normal';
    END IF;
END;
/

-- 3. Procedure (Using the Function for consistency)
CREATE OR REPLACE PROCEDURE Update_Fill_Level(p_id IN NUMBER, p_level IN NUMBER) AS
BEGIN
    UPDATE Bins 
    SET fill_level = p_level,
        -- UPDATED: Now uses your function instead of hardcoded CASE
        collection_status = Get_Fill_Status_Label(p_level) 
    WHERE bin_id = p_id;

    IF SQL%FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Bin ' || p_id || ' updated to ' || p_level || '%');
    ELSE
        DBMS_OUTPUT.PUT_LINE('Error: Bin not found.');
    END IF;
    COMMIT;
END;
/

-- 4. Trigger
CREATE OR REPLACE TRIGGER trg_bin_history
AFTER UPDATE OF fill_level ON Bins
FOR EACH ROW
BEGIN
    INSERT INTO Fill_Level_History (bin_id, recorded_level)
    VALUES (:NEW.bin_id, :NEW.fill_level);
END;
/

-- 5. View
CREATE OR REPLACE VIEW V_Bins_To_Collect AS
SELECT bin_id, area_name, location_address, fill_level, collection_status
FROM Bins
WHERE fill_level >= 75;
