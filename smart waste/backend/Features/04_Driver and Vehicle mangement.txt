SET SERVEROUTPUT ON;

-- ==========================================
-- 1. NORMALIZED TABLE STRUCTURE (3NF/1NF)
-- ==========================================
DROP TABLE Driver_Phones CASCADE CONSTRAINTS; 
DROP TABLE Drivers CASCADE CONSTRAINTS;
DROP TABLE Vehicles CASCADE CONSTRAINTS;

-- Feature 4.1: Vehicle Master
CREATE TABLE Vehicles (
    vehicle_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    registration_number VARCHAR2(20) UNIQUE NOT NULL,
    vehicle_type VARCHAR2(50),
    capacity_kg NUMBER(10,2),
    status VARCHAR2(20) DEFAULT 'Available' 
        CHECK (status IN ('Available', 'Assigned', 'Maintenance'))
);

-- Feature 4.2: Driver Master
CREATE TABLE Drivers (
    driver_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name VARCHAR2(100) NOT NULL,
    status VARCHAR2(20) DEFAULT 'Available' 
        CHECK (status IN ('Available', 'Assigned', 'On Leave'))
);

-- NORMALIZATION: Handling Multi-valued Phone Numbers (1:M Relationship)
CREATE TABLE Driver_Phones (
    driver_id NUMBER REFERENCES Drivers(driver_id) ON DELETE CASCADE,
    phone_number VARCHAR2(15),
    phone_type VARCHAR2(20) CHECK (phone_type IN ('Work', 'Personal', 'Emergency')),
    PRIMARY KEY (driver_id, phone_number)
);

-- ==========================================
-- 2. PL/SQL PROCEDURES (Implicit Cursors)
-- ==========================================

-- Procedure to Register a Driver (Feature 4.2)
CREATE OR REPLACE PROCEDURE Register_Driver (p_name IN VARCHAR2) AS
BEGIN
    INSERT INTO Drivers (name) VALUES (p_name);
    
    IF SQL%FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Success: Driver ' || p_name || ' registered.');
    END IF;
    COMMIT;
END;
/

-- Procedure to Add a Phone Number (Normalization Logic)
CREATE OR REPLACE PROCEDURE Add_Driver_Phone (p_driver_id IN NUMBER, p_phone IN VARCHAR2, p_type IN VARCHAR2) AS
BEGIN
    INSERT INTO Driver_Phones (driver_id, phone_number, phone_type)
    VALUES (p_driver_id, p_phone, p_type);

    IF SQL%FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Phone added for Driver ID: ' || p_driver_id);
    END IF;
    COMMIT;
END;
/

-- Procedure to Register a Vehicle (Feature 4.1)
CREATE OR REPLACE PROCEDURE Register_Vehicle (p_reg_no IN VARCHAR2, p_type IN VARCHAR2, p_cap IN NUMBER) AS
BEGIN
    INSERT INTO Vehicles (registration_number, vehicle_type, capacity_kg)
    VALUES (p_reg_no, p_type, p_cap);

    IF SQL%FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Success: Vehicle ' || p_reg_no || ' registered.');
    END IF;
    COMMIT;
END;
/

-- ADDED: Procedure to Update Status (Feature 4.5)
-- Useful for marking vehicles for maintenance or drivers on leave
CREATE OR REPLACE PROCEDURE Update_Resource_Status (
    p_resource_type IN VARCHAR2, -- 'DRIVER' or 'VEHICLE'
    p_id IN NUMBER,
    p_new_status IN VARCHAR2
) AS
BEGIN
    IF UPPER(p_resource_type) = 'VEHICLE' THEN
        UPDATE Vehicles SET status = p_new_status WHERE vehicle_id = p_id;
    ELSIF UPPER(p_resource_type) = 'DRIVER' THEN
        UPDATE Drivers SET status = p_new_status WHERE driver_id = p_id;
    END IF;

    IF SQL%FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Status updated for ' || p_resource_type || ' ID: ' || p_id);
    END IF;
    COMMIT;
END;
/

-- ==========================================
-- 3. VIEWS (Normal SQL)
-- ==========================================

-- View for Contact List (Joins normalized phone data)
CREATE OR REPLACE VIEW V_Driver_Contact_List AS
SELECT d.driver_id, d.name, dp.phone_number, dp.phone_type, d.status
FROM Drivers d
LEFT JOIN Driver_Phones dp ON d.driver_id = dp.driver_id;

-- View for Available Resources (Feature 4.3)
CREATE OR REPLACE VIEW V_Available_Resources AS
SELECT 'Vehicle' as type, registration_number as identifier, status FROM Vehicles WHERE status = 'Available'
UNION ALL
SELECT 'Driver' as type, name as identifier, status FROM Drivers WHERE status = 'Available';
