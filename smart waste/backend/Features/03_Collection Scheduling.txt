SET SERVEROUTPUT ON;

-- ==========================================
-- 1. TABLE STRUCTURE
-- ==========================================
DROP TABLE Schedules CASCADE CONSTRAINTS;

CREATE TABLE Schedules (
    schedule_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    -- UPDATED: Added ON DELETE CASCADE for easier testing cleanup
    route_id NUMBER REFERENCES Routes(route_id) ON DELETE CASCADE,
    -- UPDATED: Added explicit references to normalized tables
    driver_id NUMBER REFERENCES Drivers(driver_id) ON DELETE SET NULL,
    vehicle_id NUMBER REFERENCES Vehicles(vehicle_id) ON DELETE SET NULL,
    schedule_date DATE NOT NULL,
    start_time VARCHAR2(10), 
    status VARCHAR2(20) DEFAULT 'Scheduled' 
        CHECK (status IN ('Scheduled', 'In Progress', 'Completed', 'Cancelled'))
);

-- ==========================================
-- 2. PL/SQL FUNCTION
-- ==========================================
CREATE OR REPLACE FUNCTION Get_Schedule_Count(p_date IN DATE) 
RETURN NUMBER AS
    v_count NUMBER;
BEGIN
    SELECT COUNT(*) INTO v_count 
    FROM Schedules 
    WHERE TRUNC(schedule_date) = TRUNC(p_date);
    
    RETURN v_count;
END;
/

-- ==========================================
-- 3. PL/SQL PROCEDURES (Implicit Cursors)
-- ==========================================

CREATE OR REPLACE PROCEDURE Create_Collection_Schedule (
    p_route_id IN NUMBER,
    p_driver_id IN NUMBER,
    p_vehicle_id IN NUMBER,
    p_date IN DATE,
    p_time IN VARCHAR2
) AS
BEGIN
    INSERT INTO Schedules (route_id, driver_id, vehicle_id, schedule_date, start_time)
    VALUES (p_route_id, p_driver_id, p_vehicle_id, p_date, p_time);

    -- CONCEPT: Implicit Cursor (Requirement 3.1)
    IF SQL%FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Success: Schedule created for Route ' || p_route_id);
    END IF;
    COMMIT;
END;
/

CREATE OR REPLACE PROCEDURE Update_Schedule_Status (
    p_sched_id IN NUMBER,
    p_new_status IN VARCHAR2
) AS
BEGIN
    UPDATE Schedules 
    SET status = p_new_status 
    WHERE schedule_id = p_sched_id;
    
    IF SQL%FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Schedule ' || p_sched_id || ' updated to ' || p_new_status);
    ELSE
        DBMS_OUTPUT.PUT_LINE('Error: Schedule ID not found.');
    END IF;
    COMMIT;
END;
/

-- ==========================================
-- 4. TRIGGER (Validation Logic)
-- ==========================================
CREATE OR REPLACE TRIGGER trg_validate_schedule_date
BEFORE INSERT OR UPDATE ON Schedules
FOR EACH ROW
BEGIN
    -- Business Rule: Prevent scheduling in the past
    IF :NEW.schedule_date < TRUNC(SYSDATE) THEN
        RAISE_APPLICATION_ERROR(-20002, 'Past date scheduling not allowed.');
    END IF;
END;
/

-- ==========================================
-- 5. VIEWS (Normal SQL)
-- ==========================================

-- Today's Schedule View (Joins normalized data)
CREATE OR REPLACE VIEW V_Today_Schedules AS
SELECT 
    s.schedule_id,
    s.start_time,
    r.route_name,
    v.registration_number AS vehicle_reg,
    d.name AS driver_name,
    s.status
FROM Schedules s
JOIN Routes r ON s.route_id = r.route_id
JOIN Vehicles v ON s.vehicle_id = v.vehicle_id
JOIN Drivers d ON s.driver_id = d.driver_id
WHERE TRUNC(s.schedule_date) = TRUNC(SYSDATE);
