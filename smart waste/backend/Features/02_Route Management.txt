SET SERVEROUTPUT ON;

-- 1. Tables
DROP TABLE Route_Bins CASCADE CONSTRAINTS;
DROP TABLE Routes CASCADE CONSTRAINTS;

CREATE TABLE Routes (
    route_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    route_name VARCHAR2(100) NOT NULL,
    -- ADDED THESE: Required for Reports and Master Test
    start_location VARCHAR2(100),
    end_location VARCHAR2(100),
    route_status VARCHAR2(20) DEFAULT 'Active'
);

CREATE TABLE Route_Bins (
    route_id NUMBER REFERENCES Routes(route_id) ON DELETE CASCADE,
    -- UPDATED: Added Reference to Bins for Data Integrity
    bin_id NUMBER REFERENCES Bins(bin_id) ON DELETE CASCADE, 
    stop_order NUMBER,
    PRIMARY KEY (route_id, bin_id)
);

-- 3. Procedure (Updated with Implicit Cursor requirement)
CREATE OR REPLACE PROCEDURE Assign_Bin_To_Route(p_route_id NUMBER, p_bin_id NUMBER, p_stop NUMBER) AS
BEGIN
    INSERT INTO Route_Bins (route_id, bin_id, stop_order)
    VALUES (p_route_id, p_bin_id, p_stop);

    -- CONCEPT: Implicit Cursor (Requirement 2.1)
    IF SQL%FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Bin ' || p_bin_id || ' assigned to Route ' || p_route_id);
    END IF;
    
    COMMIT;
END;
/

-- 4. KEEP: Cursors (These are perfect)
CREATE OR REPLACE PROCEDURE Process_Route_Sequence(p_route_id IN NUMBER) AS
    CURSOR c_route IS 
        SELECT rb.stop_order, rb.bin_id, b.location_address 
        FROM Route_Bins rb
        JOIN Bins b ON rb.bin_id = b.bin_id -- Added Join to show address
        WHERE rb.route_id = p_route_id ORDER BY rb.stop_order;
BEGIN
    DBMS_OUTPUT.PUT_LINE('Starting collection for Route ' || p_route_id);
    FOR r IN c_route LOOP
        DBMS_OUTPUT.PUT_LINE('Stop: ' || r.stop_order || ' | Bin: ' || r.bin_id || ' at ' || r.location_address);
    END LOOP;
END;
/

CREATE OR REPLACE PROCEDURE Check_Route_Status(p_route_id IN NUMBER, p_level IN NUMBER) AS
    CURSOR c_check(cp_route NUMBER, cp_level NUMBER) IS 
        SELECT rb.bin_id, b.fill_level
        FROM Route_Bins rb JOIN Bins b ON rb.bin_id = b.bin_id
        WHERE rb.route_id = cp_route AND b.fill_level >= cp_level;
BEGIN
    FOR b_rec IN c_check(p_route_id, p_level) LOOP
        DBMS_OUTPUT.PUT_LINE('ALERT: Bin ' || b_rec.bin_id || ' is critical (' || b_rec.fill_level || '%)');
    END LOOP;
END;
/

-- 5. VIEW (Updated with location columns)
CREATE OR REPLACE VIEW V_Route_Summary AS
SELECT r.route_id, r.route_name, r.start_location, r.end_location,
       (SELECT COUNT(*) FROM Route_Bins rb WHERE rb.route_id = r.route_id) as bin_count
FROM Routes r;
