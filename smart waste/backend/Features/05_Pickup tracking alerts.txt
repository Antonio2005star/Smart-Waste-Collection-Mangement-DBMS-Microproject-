SET SERVEROUTPUT ON;

-- ==========================================
-- 1. TABLE STRUCTURE
-- ==========================================
DROP TABLE Pickups CASCADE CONSTRAINTS;

CREATE TABLE Pickups (
    pickup_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    -- UPDATED: Added ON DELETE CASCADE for seamless testing
    schedule_id NUMBER REFERENCES Schedules(schedule_id) ON DELETE CASCADE,
    bin_id NUMBER REFERENCES Bins(bin_id) ON DELETE CASCADE,
    status VARCHAR2(20) DEFAULT 'Pending' 
        CHECK (status IN ('Pending', 'Collected', 'Skipped', 'Failed')),
    reason VARCHAR2(255), 
    collection_time TIMESTAMP
);

-- ==========================================
-- 2. PL/SQL FUNCTION (Feature 5.5 / 6.6)
-- ==========================================
CREATE OR REPLACE FUNCTION Get_Completion_Percentage(p_schedule_id IN NUMBER) 
RETURN NUMBER AS
    v_total NUMBER;
    v_done  NUMBER;
BEGIN
    SELECT COUNT(*) INTO v_total FROM Pickups WHERE schedule_id = p_schedule_id;
    SELECT COUNT(*) INTO v_done FROM Pickups WHERE schedule_id = p_schedule_id AND status = 'Collected';
    
    -- Added NULLIF check to prevent divide-by-zero errors
    IF v_total = 0 THEN RETURN 0; END IF;
    RETURN ROUND((v_done / v_total) * 100, 2);
END;
/

-- ==========================================
-- 3. PL/SQL PROCEDURE (Implicit Cursor - Feature 5.2)
-- ==========================================
CREATE OR REPLACE PROCEDURE Update_Pickup_Status (
    p_id IN NUMBER,
    p_status IN VARCHAR2,
    p_reason IN VARCHAR2 DEFAULT NULL
) AS
BEGIN
    UPDATE Pickups 
    SET status = p_status, 
        reason = p_reason, 
        collection_time = CASE WHEN p_status = 'Collected' THEN SYSTIMESTAMP ELSE NULL END
    WHERE pickup_id = p_id;

    -- CONCEPT: Implicit Cursor (Requirement 5.2)
    IF SQL%FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Success: Pickup updated to ' || p_status);
    ELSE
        DBMS_OUTPUT.PUT_LINE('Error: Pickup record not found.');
    END IF;
    COMMIT;
END;
/

-- ==========================================
-- 4. TRIGGER (Automatic Feedback Loop)
-- ==========================================
CREATE OR REPLACE TRIGGER trg_reset_bin_on_collect
AFTER UPDATE OF status ON Pickups
FOR EACH ROW
WHEN (NEW.status = 'Collected')
BEGIN
    -- UPDATED: Resets both level and status to keep Module 1 in sync
    UPDATE Bins 
    SET fill_level = 0, 
        collection_status = 'Normal' 
    WHERE bin_id = :NEW.bin_id;
END;
/

-- ==========================================
-- 5. PROCEDURES WITH CURSOR FOR LOOPS
-- ==========================================

-- CONCEPT: Explicit Cursor FOR Loop (Feature 5.1)
CREATE OR REPLACE PROCEDURE Start_Route_Manifest(p_schedule_id IN NUMBER) AS
    CURSOR c_manifest IS 
        SELECT p.pickup_id, b.location_address, p.status
        FROM Pickups p
        JOIN Bins b ON p.bin_id = b.bin_id
        WHERE p.schedule_id = p_schedule_id;
BEGIN
    DBMS_OUTPUT.PUT_LINE('--- ROUTE MANIFEST FOR SCHEDULE #' || p_schedule_id || ' ---');
    FOR rec IN c_manifest LOOP
        DBMS_OUTPUT.PUT_LINE('Pickup ID: ' || rec.pickup_id || ' | Location: ' || rec.location_address || ' | Status: ' || rec.status);
    END LOOP;
END;
/

-- CONCEPT: Parameterized Cursor FOR Loop (Feature 5.4)
CREATE OR REPLACE PROCEDURE Alert_Unsuccessful_Pickups(p_schedule_id IN NUMBER) AS
    CURSOR c_alerts(cp_sched NUMBER) IS 
        SELECT p.bin_id, p.status, p.reason, b.location_address
        FROM Pickups p
        JOIN Bins b ON p.bin_id = b.bin_id
        WHERE p.schedule_id = cp_sched AND p.status IN ('Skipped', 'Failed');
BEGIN
    DBMS_OUTPUT.PUT_LINE('--- ACTION REQUIRED: UNSUCCESSFUL PICKUPS ---');
    FOR alert_rec IN c_alerts(p_schedule_id) LOOP
        DBMS_OUTPUT.PUT_LINE('ALERT: Bin ' || alert_rec.bin_id || ' (' || alert_rec.status || ') at ' || alert_rec.location_address);
        DBMS_OUTPUT.PUT_LINE('Reason: ' || alert_rec.reason);
    END LOOP;
END;
/

-- ==========================================
-- 6. VIEW (Normal SQL)
-- ==========================================
CREATE OR REPLACE VIEW V_Pending_Pickups AS
SELECT p.pickup_id, r.route_name, b.location_address, b.fill_level
FROM Pickups p
JOIN Schedules s ON p.schedule_id = s.schedule_id
JOIN Routes r ON s.route_id = r.route_id
JOIN Bins b ON p.bin_id = b.bin_id
WHERE p.status = 'Pending';
