-- Feature 1 & 2 & 4: Smart Dispatch Procedure (Sequence-free & NVL-free)
CREATE OR REPLACE PROCEDURE sp_run_dispatch (
    p_vehicle_id IN NUMBER,
    p_route_id   OUT NUMBER
) AS
    v_status VARCHAR2(20);
    v_seq    NUMBER := 1;
    v_next_route_id NUMBER := 1;
    v_next_stop_id NUMBER := 1;
    v_count  NUMBER;
BEGIN
    SELECT status INTO v_status FROM vehicles WHERE vehicle_id = p_vehicle_id;
    
    IF v_status = 'In Use' THEN
        RAISE_APPLICATION_ERROR(-20001, 'Selected vehicle is already In Use.');
    ELSIF v_status = 'Maintenance' THEN
        RAISE_APPLICATION_ERROR(-20002, 'Selected vehicle is Under Maintenance.');
    END IF;

    -- Manual route_id generation using COUNT/MAX check
    SELECT COUNT(*) INTO v_count FROM routes;
    IF v_count > 0 THEN
        SELECT MAX(route_id) + 1 INTO v_next_route_id FROM routes;
    END IF;
    p_route_id := v_next_route_id;

    INSERT INTO routes (route_id, vehicle_id, scheduled_date, status)
    VALUES (p_route_id, p_vehicle_id, SYSDATE, 'In Progress');

    UPDATE vehicles SET status = 'In Use' WHERE vehicle_id = p_vehicle_id;

    -- Manual stop_id generation using COUNT/MAX check
    SELECT COUNT(*) INTO v_count FROM route_stops;
    IF v_count > 0 THEN
        SELECT MAX(stop_id) INTO v_next_stop_id FROM route_stops;
    ELSE
        v_next_stop_id := 0;
    END IF;

    FOR b IN (SELECT bin_id FROM bins WHERE current_fill_level >= 80) LOOP
        v_next_stop_id := v_next_stop_id + 1;
        INSERT INTO route_stops (stop_id, route_id, bin_id, stop_sequence)
        VALUES (v_next_stop_id, p_route_id, b.bin_id, v_seq);
        v_seq := v_seq + 1;
    END LOOP;
    
    COMMIT;
END;
/

-- Feature 1 & 2: Route Completion Procedure (Sequence-free & NVL-free)
CREATE OR REPLACE PROCEDURE sp_complete_route (
    p_route_id IN NUMBER
) AS
    v_vehicle_id NUMBER;
    v_next_log_id NUMBER := 1;
    v_log_count NUMBER;
BEGIN
    SELECT vehicle_id INTO v_vehicle_id FROM routes WHERE route_id = p_route_id;

    -- Manual log_id generation using COUNT/MAX check
    SELECT COUNT(*) INTO v_log_count FROM collection_logs;
    IF v_log_count > 0 THEN
        SELECT MAX(log_id) + 1 INTO v_next_log_id FROM collection_logs;
    END IF;

    FOR s IN (
        SELECT rs.bin_id, b.current_fill_level 
        FROM route_stops rs 
        JOIN bins b ON rs.bin_id = b.bin_id 
        WHERE rs.route_id = p_route_id
    ) LOOP
        INSERT INTO collection_logs (log_id, bin_id, vehicle_id, collected_at, fill_level_before_collection)
        VALUES (v_next_log_id, s.bin_id, v_vehicle_id, SYSTIMESTAMP, s.current_fill_level);

        v_next_log_id := v_next_log_id + 1;

        UPDATE bins SET current_fill_level = 0 WHERE bin_id = s.bin_id;
    END LOOP;

    UPDATE routes SET status = 'Completed' WHERE route_id = p_route_id;
    UPDATE vehicles SET status = 'Available' WHERE vehicle_id = v_vehicle_id;

    COMMIT;
END;
/

-- Feature 3: IoT Sensor Simulation Procedure
CREATE OR REPLACE PROCEDURE sp_simulate_sensor_spike AS
BEGIN
    UPDATE bins 
    SET current_fill_level = LEAST(100, current_fill_level + ROUND(DBMS_RANDOM.VALUE(20, 50)))
    WHERE current_fill_level < 80;
    COMMIT;
END;
/

-- Feature 5: Historical Collection Analytics View (NVL replaced with CASE)
CREATE OR REPLACE VIEW vw_collection_analytics AS
SELECT 
    v.vehicle_id,
    v.registration_number,
    COUNT(DISTINCT cl.log_id) AS total_bins_collected,
    CASE 
        WHEN SUM(cl.fill_level_before_collection) IS NULL THEN 0 
        ELSE SUM(cl.fill_level_before_collection) 
    END AS total_waste_volume_cleared,
    MAX(cl.collected_at) AS last_collection_time
FROM vehicles v
LEFT JOIN collection_logs cl ON v.vehicle_id = cl.vehicle_id
GROUP BY v.vehicle_id, v.registration_number;
/
