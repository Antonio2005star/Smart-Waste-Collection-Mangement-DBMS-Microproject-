
-- 1. Bins Table (Master)
CREATE TABLE bins (
    bin_id INT PRIMARY KEY AUTO_INCREMENT,
    location_name VARCHAR(150) NOT NULL,
    latitude DECIMAL(10, 8) NOT NULL,
    longitude DECIMAL(11, 8) NOT NULL,
    capacity_liters INT DEFAULT 1000,
    current_fill_level INT DEFAULT 0
);

-- 2. Vehicles Table (Master)
CREATE TABLE vehicles (
    vehicle_id INT PRIMARY KEY AUTO_INCREMENT,
    registration_number VARCHAR(20) NOT NULL,
    capacity_liters INT DEFAULT 5000,
    status VARCHAR(20) DEFAULT 'Available'
);

-- 3. Routes Table (Transaction Master)
CREATE TABLE routes (
    route_id INT PRIMARY KEY AUTO_INCREMENT,
    vehicle_id INT NOT NULL,
    scheduled_date DATE NOT NULL,
    status VARCHAR(20) DEFAULT 'Scheduled',
    FOREIGN KEY (vehicle_id) REFERENCES vehicles(vehicle_id) ON DELETE CASCADE
);

-- 4. Route Stops Table (Junction Table: Routes <-> Bins)
CREATE TABLE route_stops (
    stop_id INT PRIMARY KEY AUTO_INCREMENT,
    route_id INT NOT NULL,
    bin_id INT NOT NULL,
    stop_sequence INT NOT NULL,
    FOREIGN KEY (route_id) REFERENCES routes(route_id) ON DELETE CASCADE,
    FOREIGN KEY (bin_id) REFERENCES bins(bin_id) ON DELETE CASCADE
);

-- 5. Collection Logs Table (Audit History)
CREATE TABLE collection_logs (
    log_id INT PRIMARY KEY AUTO_INCREMENT,
    bin_id INT NOT NULL,
    vehicle_id INT NOT NULL,
    collected_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    fill_level_before_collection INT NOT NULL,
    FOREIGN KEY (bin_id) REFERENCES bins(bin_id) ON DELETE CASCADE,
    FOREIGN KEY (vehicle_id) REFERENCES vehicles(vehicle_id) ON DELETE CASCADE
);
