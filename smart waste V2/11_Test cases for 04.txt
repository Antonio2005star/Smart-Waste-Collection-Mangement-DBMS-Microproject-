--Test A: Verify Smart Dispatch (proc_smart_dispatch)
-- Run smart dispatch for Vehicle 1
EXEC proc_smart_dispatch(1);

-- Verify Vehicle 1 status changed to 'In Use'
SELECT vehicle_id, registration_number, capacity_liters, status 
FROM vehicles 
WHERE vehicle_id = 1;

-- Verify new route generated for Vehicle 1
SELECT route_id, vehicle_id, scheduled_date, status 
FROM routes 
WHERE vehicle_id = 1;

--Test B: Verify Auto-Trigger on Critical Spike (trg_auto_trigger_dispatch) SQL

-- Ensure Vehicle 3 is marked 'Available' to receive the trigger assignment
UPDATE vehicles SET status = 'Available' WHERE vehicle_id = 3;
COMMIT;

-- Update Bin 3 (Fort Kochi Beach) fill level to 90%
-- This automatically triggers trg_auto_trigger_dispatch
UPDATE bins 
SET current_fill_level = 90 
WHERE bin_id = 3;

COMMIT;

-- Verify Vehicle 3 was automatically dispatched and set to 'In Use'
SELECT vehicle_id, registration_number, status 
FROM vehicles 
WHERE vehicle_id = 3;

--Test C: Verify Route Completion & Auto Bin Reset (proc_complete_route) SQL

-- Complete Route 1 (logs collections and triggers bin fill resets)
EXEC proc_complete_route(1);

-- Verify bin fill levels assigned to Route 1 were automatically reset to 0%
SELECT b.bin_id, b.location_name, b.current_fill_level 
FROM bins b
JOIN route_stops rs ON b.bin_id = rs.bin_id
WHERE rs.route_id = 1;

-- Verify Route 1 status is updated to 'Completed'
SELECT route_id, vehicle_id, status FROM routes WHERE route_id = 1;

-- Verify Vehicle 1 status is returned to 'Available'
SELECT vehicle_id, registration_number, status FROM vehicles WHERE vehicle_id = 1;
