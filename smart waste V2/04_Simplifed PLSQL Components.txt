-- A. Function: Calculate Total Waste Assigned to a Vehicle
CREATE OR REPLACE FUNCTION fn_get_vehicle_load (
    p_vehicle_id IN NUMBER
) RETURN NUMBER IS
    v_total_fill NUMBER := 0;
BEGIN
    SELECT SUM(b.current_fill_level)
    INTO v_total_fill
    FROM routes r
    JOIN route_stops rs ON r.route_id = rs.route_id
    JOIN bins b ON rs.bin_id = b.bin_id
    WHERE r.vehicle_id = p_vehicle_id 
      AND r.status = 'In Progress';
      
    IF v_total_fill IS NULL THEN
        v_total_fill := 0;
    END IF;
    
    RETURN v_total_fill;
END;
/

-- B. Procedure & Cursor: Assign High-Fill Bins to a Route (Sequence & NVL/COALESCE Free)
CREATE OR REPLACE PROCEDURE proc_assign_high_bins (
    p_vehicle_id IN NUMBER
) IS
    CURSOR c_high_bins IS
        SELECT bin_id, location_name
        FROM bins
        WHERE current_fill_level >= 80;
        
    v_bin_id        NUMBER;
    v_location_name VARCHAR2(150);
    v_route_id      NUMBER := 1;
    v_stop_id       NUMBER := 1;
    v_stop_no       NUMBER := 1;
    v_count         NUMBER;
BEGIN
    -- Manual route_id generation using COUNT/MAX check
    SELECT COUNT(*) INTO v_count FROM routes;
    IF v_count > 0 THEN
        SELECT MAX(route_id) + 1 INTO v_route_id FROM routes;
    END IF;

    INSERT INTO routes (route_id, vehicle_id, scheduled_date, status)
    VALUES (v_route_id, p_vehicle_id, SYSDATE, 'In Progress');

    -- Manual stop_id generation using COUNT/MAX check
    SELECT COUNT(*) INTO v_count FROM route_stops;
    IF v_count > 0 THEN
        SELECT MAX(stop_id) INTO v_stop_id FROM route_stops;
    ELSE
        v_stop_id := 0;
    END IF;

    OPEN c_high_bins;
    LOOP
        FETCH c_high_bins INTO v_bin_id, v_location_name;
        EXIT WHEN c_high_bins%NOTFOUND;
        
        v_stop_id := v_stop_id + 1;

        INSERT INTO route_stops (stop_id, route_id, bin_id, stop_sequence)
        VALUES (v_stop_id, v_route_id, v_bin_id, v_stop_no);
        
        v_stop_no := v_stop_no + 1;
    END LOOP;
    CLOSE c_high_bins;
END;
/

-- C. Smart Dispatch Procedure without COMMIT
CREATE OR REPLACE PROCEDURE proc_smart_dispatch (
    p_vehicle_id IN NUMBER
) IS
    v_current_load NUMBER := 0;
    v_max_capacity NUMBER := 0;
BEGIN
    v_current_load := fn_get_vehicle_load(p_vehicle_id);
    
    SELECT capacity_liters 
    INTO v_max_capacity 
    FROM vehicles 
    WHERE vehicle_id = p_vehicle_id;

    IF v_current_load < v_max_capacity THEN
        proc_assign_high_bins(p_vehicle_id);
        
        UPDATE vehicles 
        SET status = 'In Use' 
        WHERE vehicle_id = p_vehicle_id;
    END IF;
END;
/

-- D. Statement-Level Auto-Trigger for Critical Bin Spikes
CREATE OR REPLACE TRIGGER trg_auto_trigger_dispatch
AFTER UPDATE OF current_fill_level ON bins
DECLARE
    v_available_vehicle_id NUMBER;
    v_critical_count       NUMBER := 0;
BEGIN
    SELECT COUNT(*) 
    INTO v_critical_count 
    FROM bins 
    WHERE current_fill_level >= 80;

    IF v_critical_count > 0 THEN
        SELECT vehicle_id 
        INTO v_available_vehicle_id
        FROM vehicles
        WHERE status = 'Available'
        FETCH FIRST 1 ROWS ONLY;

        proc_smart_dispatch(v_available_vehicle_id);
    END IF;
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        NULL;
END;
/

-- E. Trigger: Reset Bin Level to 0 After Pickup
CREATE OR REPLACE TRIGGER trg_reset_bin_level
AFTER INSERT ON collection_logs
FOR EACH ROW
BEGIN
    UPDATE bins
    SET current_fill_level = 0
    WHERE bin_id = :NEW.bin_id;
END;
/

COMMIT;
