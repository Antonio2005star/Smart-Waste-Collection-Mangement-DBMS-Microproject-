-- ============================================================
-- SMART WASTE COLLECTION MANAGEMENT
-- VIEWS, PL/SQL & TRIGGERS
-- ============================================================

SET SERVEROUTPUT ON
SET LINESIZE 150
SET PAGESIZE 50
SET WRAP OFF

COLUMN street_address FORMAT A25
COLUMN zone_name FORMAT A20
COLUMN status FORMAT A15
COLUMN fill_level_percent FORMAT 990.99


-- ============================================================
-- 1. VIEW
-- Critical bins with fill level of 80% or higher
-- ============================================================

CREATE OR REPLACE VIEW vw_critical_bins AS
SELECT
    b.bin_id,
    l.street_address,
    z.zone_name,
    b.fill_level_percent,
    b.status
FROM Bins b
JOIN Locations l ON b.location_id = l.location_id
JOIN Zones z ON l.zone_id = z.zone_id
WHERE b.fill_level_percent >= 80;


-- Test the view

SELECT *
FROM vw_critical_bins;


-- ============================================================
-- 2. TRIGGER
-- Display an alert when a bin crosses 80% fill level
-- ============================================================

CREATE OR REPLACE TRIGGER trg_bin_overflow_alert
BEFORE UPDATE OF fill_level_percent ON Bins
FOR EACH ROW
BEGIN
    IF :NEW.fill_level_percent >= 80
       AND :OLD.fill_level_percent < 80 THEN

        DBMS_OUTPUT.PUT_LINE(
            '[ALERT] Bin ' || :NEW.bin_id ||
            ' exceeded 80% fill level.'
        );

        DBMS_OUTPUT.PUT_LINE(
            'New Fill Level: ' ||
            :NEW.fill_level_percent || '%'
        );

    END IF;
END;
/


-- Test the trigger

UPDATE Bins
SET fill_level_percent = 88
WHERE bin_id = 502;

COMMIT;


-- ============================================================
-- 3. PL/SQL PROCEDURE
-- Complete a bin collection
-- ============================================================

CREATE OR REPLACE PROCEDURE prc_complete_bin_pickup (
    p_bin_id IN NUMBER,
    p_driver_id IN NUMBER
) AS
BEGIN

    -- Reset bin after collection
    UPDATE Bins
    SET fill_level_percent = 0,
        status = 'Active'
    WHERE bin_id = p_bin_id;

    -- Make driver available
    UPDATE Drivers
    SET status = 'Available'
    WHERE driver_id = p_driver_id;

    -- Save the transaction
    COMMIT;

    DBMS_OUTPUT.PUT_LINE(
        '[SUCCESS] Bin ' || p_bin_id ||
        ' cleared successfully.'
    );

EXCEPTION
    WHEN OTHERS THEN

        -- Undo changes if an error occurs
        ROLLBACK;

        DBMS_OUTPUT.PUT_LINE(
            '[ERROR] Pickup failed. Changes rolled back.'
        );

END;
/


-- Test the procedure

EXEC prc_complete_bin_pickup(501, 2);


-- Verify the result

SELECT
    bin_id,
    fill_level_percent,
    status
FROM Bins
WHERE bin_id = 501;
